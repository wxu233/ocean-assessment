    15	def connect(path=None) -> sqlite3.Connection:
    16	    """Open a connection in autocommit mode; callers BEGIN explicitly."""
    17	    path = Path(path or db_path())
    18	    path.parent.mkdir(parents=True, exist_ok=True)
    19	    conn = sqlite3.connect(path, autocommit=True, check_same_thread=False, timeout=30)
    20	    conn.row_factory = sqlite3.Row
    21	    conn.execute("PRAGMA foreign_keys=ON")
    22	    conn.execute("PRAGMA journal_mode=WAL")
    23	    conn.execute("PRAGMA busy_timeout=30000")
    24	    return conn

    45	
    46	# Whitelists: request values map to fixed SQL fragments; nothing from the request is
    47	# interpolated. `issued_date` sorts on the stored tenant-local columns (A5).
    48	SORT_COLUMNS = {
    49	    "issued_date": ("issued_local_date", "issued_local_time"),
    50	    "learner_name": ("learner_name",),
    51	    "course_title": ("course_title",),
    52	    "serial": ("serial",),
    53	    "status": ("status",),
    54	}
    55	SORT_DIRS = {"asc": "ASC", "desc": "DESC"}
    56	EXPORT_COLS = "learner_name, course_title, serial, issued_local_date"
    57	
    58	
    59	def _like_prefix(q: str) -> str:
    60	    return q.replace("\\", "\\\\").replace("%", "\\%").replace("_", "\\_") + "%"
    61	
    62	
    63	def _cert_filter(tenant_id: int, q: str | None) -> tuple[str, list]:
    64	    """WHERE clause: tenant, plus `q` as a case-insensitive prefix of the learner name or
    65	    the serial (an exact serial is its own prefix). LIKE ignores case for ASCII only."""
    66	    if not q:
    67	        return "tenant_id = ?", [tenant_id]
    68	    pattern = _like_prefix(q)
    69	    return (
    70	        "tenant_id = ? AND (learner_name LIKE ? ESCAPE '\\' OR serial LIKE ? ESCAPE '\\')",
    71	        [tenant_id, pattern, pattern],
    72	    )
    73	
    74	
    75	def build_cert_query(
    76	    tenant_id: int, *, q: str | None, sort: str, direction: str, columns: str
    77	) -> tuple[str, list]:
    78	    """The one SELECT shared by the list and the export, so both return the same rows
    79	    in the same order. Ties are always broken by id ASC. Unknown sort/dir raise KeyError."""
    80	    d = SORT_DIRS[direction]
    81	    order = ", ".join(f"{col} {d}" for col in SORT_COLUMNS[sort]) + ", id ASC"
    82	    where, params = _cert_filter(tenant_id, q)
    83	    return f"SELECT {columns} FROM certificates WHERE {where} ORDER BY {order}", params
    84	
    85	
    86	def list_certificates(
    87	    conn: sqlite3.Connection, tenant_id: int, *, q: str | None, sort: str, direction: str,
    88	    page: int, page_size: int,
    89	) -> dict:
    90	    where, params = _cert_filter(tenant_id, q)
    91	    sql, sel_params = build_cert_query(tenant_id, q=q, sort=sort, direction=direction,
    92	                                       columns=CERT_SUMMARY_COLS)
    93	    # Count and page in one read transaction so `total` matches `items`.
    94	    conn.execute("BEGIN")
    95	    try:
    96	        total = conn.execute(
    97	            f"SELECT COUNT(*) FROM certificates WHERE {where}", params).fetchone()[0]
    98	        rows = conn.execute(f"{sql} LIMIT ? OFFSET ?",
    99	                            [*sel_params, page_size, (page - 1) * page_size]).fetchall()
   100	    finally:
   101	        conn.execute("COMMIT")
   102	    return {"items": [dict(r) for r in rows], "total": total, "page": page,
   103	            "page_size": page_size}
   104	
   105	
   106	def iter_export_rows(
   107	    conn: sqlite3.Connection, tenant_id: int, *, q: str | None, sort: str, direction: str,
   108	    batch_size: int = 2000,
   109	):
   110	    """Yield batches of export rows from one read transaction (a single WAL snapshot),
   111	    so a long export sees a consistent view and never blocks writers or the list."""
   112	    sql, params = build_cert_query(tenant_id, q=q, sort=sort, direction=direction,
   113	                                   columns=EXPORT_COLS)
   114	    conn.execute("BEGIN")
   115	    cur = conn.cursor()
   116	    try:
   117	        cur.execute(sql, params)
   118	        while rows := cur.fetchmany(batch_size):
   119	            yield rows
   120	    finally:
   121	        cur.close()  # finalize the statement first, even if the client went away mid-stream
   122	        conn.execute("COMMIT")

    14	FORMULA_PREFIXES = ("=", "+", "-", "@", "\t", "\r")
    15	
    16	
    17	def _safe_cell(value: str) -> str:
    18	    """Stop spreadsheets from running a cell as a formula (CSV injection, per OWASP):
    19	    a leading quote makes Excel/Sheets show it as text. Stored data is unchanged."""
    20	    return "'" + value if value.startswith(FORMULA_PREFIXES) else value
    21	
    22	
    23	def _csv_chunks(tenant_id: int, q: str | None, sort: str, direction: str):
    24	    """Sync generator: Starlette runs it in the threadpool, so a long export never
    25	    blocks the event loop. It owns its connection, separate from the request's."""
    26	    conn = db.connect()
    27	    try:
    28	        buf = io.StringIO()
    29	        writer = csv.writer(buf, lineterminator="\r\n")
    30	        writer.writerow(CSV_HEADER)
    31	        for rows in queries.iter_export_rows(conn, tenant_id, q=q, sort=sort,
    32	                                             direction=direction):
    33	            writer.writerows([_safe_cell(v) for v in r] for r in rows)
    34	            yield buf.getvalue()
    35	            buf.seek(0)
    36	            buf.truncate()
    37	        if buf.tell():  # header only, when nothing matched
    38	            yield buf.getvalue()
    39	    finally:
    40	        conn.close()
    41

     1	"""Bulk issue: one BEGIN IMMEDIATE transaction per request, idempotent per (tenant, key)."""
     2	
     3	import hashlib
     4	import json
     5	import sqlite3
     6	from datetime import datetime, timezone

    10	# sqlite3's message when the certificate UNIQUE(tenant_id, course_id, learner_id) fails.
    11	DUPLICATE_CERT = "certificates.tenant_id, certificates.course_id, certificates.learner_id"
    12	
    13	
    14	class KeyReused(Exception):
    15	    """The idempotency key exists in this tenant with a different request."""
    16	
    17	
    18	class CourseNotFound(Exception):
    19	    """The course doesn't exist or belongs to another tenant."""
    20	
    21	
    22	def request_hash(body: dict) -> str:
    23	    canonical = json.dumps(body, sort_keys=True, separators=(",", ":"))
    24	    return hashlib.sha256(canonical.encode()).hexdigest()
    25	
    26	
    27	def run_bulk_issue(
    28	    conn: sqlite3.Connection,
    29	    *,
    30	    tenant_id: int,
    31	    admin_id: int,
    32	    key: str,
    33	    body: dict,
    34	) -> str:
    35	    """Run (or replay) a bulk issue and return the BulkResult JSON string.
    36	
    37	    BEGIN IMMEDIATE takes the write lock before the key lookup, so concurrent requests
    38	    with the same key serialize here: the first runs the operation, later ones replay it.
    39	    """
    40	    req_hash = request_hash(body)
    41	    course_id = body["course_id"]
    42	    conn.execute("BEGIN IMMEDIATE")
    43	    try:
    44	        existing = conn.execute(
    45	            "SELECT request_hash, response_json FROM bulk_operations "
    46	            "WHERE tenant_id = ? AND idempotency_key = ?",
    47	            (tenant_id, key),
    48	        ).fetchone()
    49	        if existing is not None:
    50	            conn.execute("ROLLBACK")
    51	            if existing["request_hash"] != req_hash:
    52	                raise KeyReused()
    53	            return existing["response_json"]
    54	
    55	        # Ownership check before anything is written.
    56	        course = conn.execute(
    57	            "SELECT id FROM courses WHERE id = ? AND tenant_id = ?", (course_id, tenant_id)
    58	        ).fetchone()
    59	        if course is None:
    60	            conn.execute("ROLLBACK")
    61	            raise CourseNotFound()
    62	
    63	        now = datetime.now(timezone.utc)
    64	        # response_json is filled in before COMMIT; the row is never visible without it.
    65	        op_id = conn.execute(
    66	            "INSERT INTO bulk_operations (tenant_id, admin_id, idempotency_key, request_hash, "
    67	            "course_id, status, response_json, created_at) "
    68	            "VALUES (?, ?, ?, ?, ?, 'completed', '', ?)",
    69	            (tenant_id, admin_id, key, req_hash, course_id, utc_iso(now)),
    70	        ).lastrowid
    71	
    72	        results = [_issue_one(conn, tenant_id, admin_id, course_id, op_id, row, now)
    73	                   for row in _enrollments(conn, tenant_id, course_id)]
    74	        summary = {o: sum(r["outcome"] == o for r in results)
    75	                   for o in ("issued", "skipped", "failed")}
    76	        response_json = json.dumps({"operation_id": op_id, "course_id": course_id,
    77	                                    "summary": summary, "results": results})
    78	        conn.execute(
    79	            "UPDATE bulk_operations SET response_json = ? WHERE id = ? AND tenant_id = ?",
    80	            (response_json, op_id, tenant_id),
    81	        )
    82	        conn.execute("COMMIT")
    83	        return response_json
    84	    except BaseException:
    85	        if conn.in_transaction:
    86	            conn.execute("ROLLBACK")
    87	        raise

    90	def _enrollments(conn: sqlite3.Connection, tenant_id: int, course_id: int) -> list[sqlite3.Row]:
    91	    return conn.execute(
    92	        "SELECT e.learner_id, l.name AS learner_name, e.status, c.id AS certificate_id "
    93	        "FROM enrollments e "
    94	        "JOIN learners l ON l.tenant_id = e.tenant_id AND l.id = e.learner_id "
    95	        "LEFT JOIN certificates c ON c.tenant_id = e.tenant_id "
    96	        "  AND c.course_id = e.course_id AND c.learner_id = e.learner_id "
    97	        "WHERE e.tenant_id = ? AND e.course_id = ? ORDER BY e.learner_id",
    98	        (tenant_id, course_id),
    99	    ).fetchall()

   102	def _issue_one(conn, tenant_id, admin_id, course_id, op_id, row, now) -> dict:
   103	    result = {"learner_id": row["learner_id"], "learner_name": row["learner_name"],
   104	              "outcome": "skipped", "reason": None, "certificate_id": None}
   105	    if row["status"] != "completed":
   106	        result["reason"] = "not_completed"
   107	        return result
   108	    if row["certificate_id"] is not None:
   109	        result["reason"] = "already_certified"
   110	        return result
   111	
   112	    conn.execute("SAVEPOINT learner")
   113	    try:
   114	        cert_id = issue_certificate(conn, tenant_id=tenant_id, course_id=course_id,
   115	                                    learner_id=row["learner_id"], issued_at_utc=now,
   116	                                    bulk_operation_id=op_id)
   117	        record_audit_event(conn, tenant_id=tenant_id, actor_admin_id=admin_id,
   118	                           action="certificate.issued", certificate_id=cert_id,
   119	                           created_at_utc=now, bulk_operation_id=op_id)
   120	    except Exception as e:
   121	        conn.execute("ROLLBACK TO learner")
   122	        conn.execute("RELEASE learner")
   123	        if isinstance(e, sqlite3.IntegrityError) and DUPLICATE_CERT in str(e):
   124	            result["reason"] = "already_certified"
   125	        else:
   126	            result["outcome"], result["reason"] = "failed", str(e) or type(e).__name__
   127	        return result
   128	    conn.execute("RELEASE learner")
   129	    result["outcome"], result["certificate_id"] = "issued", cert_id
   130	    return result

     1	"""The single certificate insert helper, used by the API, the seed and gen_large."""
     2	
     3	import secrets
     4	import sqlite3
     5	from datetime import datetime, timezone
     6	from zoneinfo import ZoneInfo
     7	
     8	SERIAL_ALPHABET = "ABCDEFGHJKMNPQRSTUVWXYZ23456789"  # no 0/O, 1/I/L
     9	SERIAL_ATTEMPTS = 5
    10	
    11	
    12	def new_serial() -> str:
    13	    def block():
    14	        return "".join(secrets.choice(SERIAL_ALPHABET) for _ in range(4))
    15	    return f"CERT-{block()}-{block()}"
    16	
    17	
    18	def utc_iso(dt: datetime) -> str:
    19	    """'2026-01-10T13:30:00Z'. Rejects naive datetimes."""
    20	    if dt.tzinfo is None:
    21	        raise ValueError("issued_at_utc must be timezone-aware")
    22	    return dt.astimezone(timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")
    23	
    24	
    25	def local_date_time(dt: datetime, tz_name: str) -> tuple[str, str]:
    26	    """(issued_local_date, issued_local_time) in the tenant's timezone."""
    27	    if dt.tzinfo is None:
    28	        raise ValueError("issued_at_utc must be timezone-aware")
    29	    local = dt.astimezone(ZoneInfo(tz_name))
    30	    return local.strftime("%Y-%m-%d"), local.strftime("%Y-%m-%d %H:%M")
    31	
    32	
    33	def issue_certificate(
    34	    conn: sqlite3.Connection,
    35	    *,
    36	    tenant_id: int,
    37	    course_id: int,
    38	    learner_id: int,
    39	    issued_at_utc: datetime,
    40	    bulk_operation_id: int | None = None,
    41	) -> int:
    42	    """Insert one active certificate and return its id.
    43	
    44	    The caller owns the transaction: this never BEGINs or COMMITs. A duplicate
    45	    (tenant, course, learner) raises sqlite3.IntegrityError from the UNIQUE constraint.
    46	    """
    47	    row = conn.execute(
    48	        "SELECT t.timezone, c.title AS course_title, l.name AS learner_name "
    49	        "FROM tenants t "
    50	        "JOIN courses c ON c.tenant_id = t.id AND c.id = ? "
    51	        "JOIN learners l ON l.tenant_id = t.id AND l.id = ? "
    52	        "WHERE t.id = ?",
    53	        (course_id, learner_id, tenant_id),
    54	    ).fetchone()
    55	    if row is None:
    56	        raise LookupError("course or learner not found in tenant")
    57	
    58	    local_date, local_time = local_date_time(issued_at_utc, row["timezone"])
    59	    for attempt in range(SERIAL_ATTEMPTS):
    60	        try:
    61	            cur = conn.execute(
    62	                "INSERT INTO certificates (tenant_id, course_id, learner_id, learner_name, "
    63	                "course_title, serial, status, issued_at_utc, issued_local_date, "
    64	                "issued_local_time, bulk_operation_id) "
    65	                "VALUES (?, ?, ?, ?, ?, ?, 'active', ?, ?, ?, ?)",
    66	                (tenant_id, course_id, learner_id, row["learner_name"], row["course_title"],
    67	                 new_serial(), utc_iso(issued_at_utc), local_date, local_time,
    68	                 bulk_operation_id),
    69	            )
    70	            return cur.lastrowid
    71	        except sqlite3.IntegrityError as e:
    72	            # Retry only a random serial collision; everything else propagates.
    73	            if "certificates.serial" not in str(e) or attempt == SERIAL_ATTEMPTS - 1:
    74	                raise
    75	    raise AssertionError("unreachable")

    11	    conn: sqlite3.Connection, *, tenant_id: int, admin_id: int, cert_id: int
    12	) -> dict | None:
    13	    """Revoke and return the CertDetail, or None if the id is missing or in another tenant.
    14	
    15	    Only the request whose UPDATE flips 'active' -> 'revoked' writes the audit event, so

    31	        conn.execute("COMMIT")

    34	        if conn.in_transaction:
    35	            conn.execute("ROLLBACK")
    36	        raise

    28	def row_shape(i: int, count: int) -> tuple[str, datetime]:
    29	    """(learner name, issued_at) for row i. Rows 2k and 2k+1 always tie on both."""
    30	    pair = i // 2
    31	    first = FIRST_NAMES[pair % len(FIRST_NAMES)]
    32	    last = LAST_NAMES[(pair // len(FIRST_NAMES)) % len(LAST_NAMES)]
    33	    slots = max(1, count // 8)
    34	    return f"{first} {last}", BASE_UTC + timedelta(minutes=SLOT_MINUTES * (pair % slots))
